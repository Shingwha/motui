```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""斐波那契数列 Fibonacci sequence."""

from typing import List


def fibonacci(n: int) -> List[int]:
    """生成前 n 个斐波那契数。Generate first n Fibonacci numbers."""
    seq = [0, 1]
    while len(seq) < n:
        seq.append(seq[-1] + seq[-2])
    return seq[:n]


if __name__ == "__main__":
    print(f"Result 结果: {fibonacci(10)}")
```
